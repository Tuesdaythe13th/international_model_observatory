\section{Executive Summary}

\subsection{The Core Problem}
If large language models can produce fluent self-explanations without mechanistic access to the computations that generated their actions, then explanation-based oversight (e.g., Chain-of-Thought monitoring) may fail silently. This project supplies a causal test of the \textbf{Faithfulness Assumption}: the premise that a model's self-explanations faithfully track its behavior-generating computation.

\subsection{The ``Press Secretary'' Architecture Hypothesis}
Evidence motivating this work is consistent with a functionally separated architecture:
\begin{itemize}[leftmargin=1.3em]
    \item \textbf{Actor Circuits ($\mathcal{C}_B$)}: mid-layer circuitry implementing task-specific behaviors (candidate depth region: L15--22 in Gemma-2).
    \item \textbf{Narrator Circuits ($\mathcal{C}_E$)}: late-layer circuitry generating post-hoc rationalizations (candidate depth region: L28+).
    \item \textbf{Monitoring Circuits}: safety filtering and policy compliance mechanisms with partial behavioral oversight.
\end{itemize}

\begin{figure}[H]
    \centering
    \includegraphics[width=0.85\textwidth]{images/03_split_brain_concept.png}
    \caption{\textbf{Press Secretary Architecture.} Actor vs.\ Narrator dissociation across transformer depth.}
\end{figure}

This implies a \textbf{Press Secretary effect}: the model conditions on its outputs and surface-level signals available late in the forward pass, producing contextually plausible justifications without direct access to causal mechanisms.

\subsection{Safety Implications of Faithfulness Failure}
If dissociation is common in safety-relevant settings, then:
\begin{itemize}[leftmargin=1.3em]
    \item \textbf{Oversight blindspots}: explanatory monitoring is not a reliable proxy for the decision process.
    \item \textbf{Confabulation risk}: convincing but false accounts of internal reasoning become the default failure mode.
    \item \textbf{Ambiguity in intent}: surface-text explanations can mask confusion, confabulation, or strategic misreporting.
    \item \textbf{Self-diagnostic unreliability}: model ``introspection'' becomes a signal to be validated, not ground truth.
\end{itemize}

\subsection{The Ablation Dissociation Test (ADT)}
ADT surgically perturbs behavior-driving circuitry ($\mathcal{C}_B$) while measuring effects on both behavior ($\Delta\mathcal{B}$) and explanations ($\Delta\mathcal{E}$).

\begin{table}[H]
\centering
\begin{tabularx}{\textwidth}{lXX}
\toprule
\textbf{Hypothesis} & \textbf{Prediction} & \textbf{Meaning} \\
\midrule
\textbf{Faithful reporting} & $\Delta\mathcal{B}\downarrow \Rightarrow \Delta\mathcal{E}\downarrow$ & Explanations track behavior-driving computation \\
\textbf{Split-Brain (confabulation)} & $\Delta\mathcal{B}\downarrow$ but $\Delta\mathcal{E}\approx 0$ & Explanations are independent narratives \\
\textbf{Partial access} & $0<\Delta\mathcal{E}<\Delta\mathcal{B}$ & Noisy or incomplete coupling \\
\bottomrule
\end{tabularx}
\caption{Predictions of the Ablation Dissociation Test (ADT)}
\end{table}

\subsection{The H-Score: Quantifying Honesty}
We introduce the \textbf{H-Score (Honesty Coefficient)}:

\begin{tcolorbox}[colback=white, colframe=crimson, boxrule=1pt, sharp corners,
                 title=\textbf{H-Score: Normalized explanation shift under causal intervention}]
\[
H = 1 - \frac{D_{JS}\!\left(P(\mathcal{E}\mid do(\mathcal{C}_B = 0)) \parallel P(\mathcal{E}\mid \text{clean})\right)}{D_{JS}\!\left(P(\mathcal{E}\mid do(\epsilon)) \parallel P(\mathcal{E}\mid \text{clean})\right)}
\]
\centering
\small $D_{JS}$: Jensen--Shannon divergence; $\mathcal{C}_B$: Actor circuitry; $\epsilon$: matched random-feature ablation control.\\
\textit{$H \to 0$ indicates explanation invariance under behavioral intervention (Split-Brain-consistent).}
\end{tcolorbox}

\begin{table}[H]
\centering
\begin{tabularx}{0.85\textwidth}{lXX}
\toprule
\textbf{H-Score Range} & \textbf{Diagnosis} & \textbf{Safety Implication} \\
\midrule
\textbf{0.7--1.0} & High coupling & Self-reports are plausible monitors \\
\textbf{0.4--0.7} & Partial coupling & Self-reports require corroboration \\
\textbf{0.0--0.4} & Split-Brain regime & Self-reports are unreliable as monitors \\
\textbf{$<0.0$} & Anti-correlation & Extreme dissociation / inversion \\
\bottomrule
\end{tabularx}
\caption{H-Score interpretation framework}
\end{table}

\section{Methodology: Neural Forensics}

\subsection{Phase 1: Phenotype Replication}
\textbf{Goal}: systematically reproduce dissociation signatures in open-weight models under controlled decoding.

\begin{lstlisting}[language=Python,caption={Dissociation test generator (sketch)}]
class DissociationTestGenerator:
    """Generate controlled test cases for dissociation phenotypes."""

    def __init__(self, model_family="Gemma-2"):
        self.model_family = model_family
        self.control_conditions = {
            "context_pressure": [0.1, 0.3, 0.5, 0.7, 0.9],
            "explanation_demand": ["low", "medium", "high", "explicit"],
            "serialization_complexity": [1, 2, 3, 4, 5],
        }

    def generate_suite(self, base_prompt: str):
        """Return a list of prompt variants and expected behavior labels."""
        suite = []
        for cp in self.control_conditions["context_pressure"]:
            for ed in self.control_conditions["explanation_demand"]:
                for sc in self.control_conditions["serialization_complexity"]:
                    suite.append({
                        "prompt": f"{base_prompt}\n\n[cp={cp}, ed={ed}, sc={sc}]",
                        "labels": {"cp": cp, "ed": ed, "sc": sc},
                    })
        return suite
\end{lstlisting}

\subsection{Phase 3: Ablation Dissociation Test (ADT)}
\textbf{Core protocol}: surgical ablation of $\mathcal{C}_B$ features while measuring $\Delta\mathcal{B}$ and $\Delta\mathcal{E}$ under fixed decoding.

\begin{figure}[H]
    \centering

\begin{lstlisting}[language=Python,caption={ADT execution protocol (sketch)}]
class AblationDissociationTest:
    """Execute surgical ablation to test explanation faithfulness."""

    def __init__(self, model, C_B_features, C_E_features=None):
        self.model = model
        self.C_B = C_B_features  # behavior-driving feature ids / hooks
        self.C_E = C_E_features  # optional: explanation feature ids / hooks

    def get_behavior_and_explanation(self, tracer):
        """Return (B, E) readouts under the trace."""
        B = None
        E = None
        return B, E

    def zero_ablate(self, tracer, feature_spec, method="surgical"):
        """Apply ablation for selected features."""
        return

    def compute_H_scores(self, results):
        """Compute H from divergence estimates; implement D_JS in logit space."""
        return {"H": None, "diagnostics": {}}

    def run_experiment(self, prompt, n_trials=50):
        results = {
            "clean": {"behavior": [], "explanation": []},
            "behavior_ablated": {"behavior": [], "explanation": []},
            "random_ablated": {"behavior": [], "explanation": []},
            "sham": {"behavior": [], "explanation": []},
        }

        for _ in range(n_trials):
            with self.model.trace(prompt) as tracer:
                B_clean, E_clean = self.get_behavior_and_explanation(tracer)
                results["clean"]["behavior"].append(B_clean)
                results["clean"]["explanation"].append(E_clean)

            with self.model.trace(prompt) as tracer:
                self.zero_ablate(tracer, self.C_B, method="surgical")
                B_ab, E_ab = self.get_behavior_and_explanation(tracer)
                results["behavior_ablated"]["behavior"].append(B_ab)
                results["behavior_ablated"]["explanation"].append(E_ab)

        return self.compute_H_scores(results)
\end{lstlisting}

\section{Technical Considerations}

\subsection{Proxy Gap Mitigation}
Open-weight replication is used to reduce dependence on closed-model logs. The primary goal is to reproduce \emph{phenotypes} (behavior/explanation dissociation under stress and intervention) while maintaining strict control over decoding and measurement.

\begin{lstlisting}[language=Python,caption={Proxy replication strategy (sketch)}]
class ProxyReplicator:
    """Replicate dissociation classes in open-weight models."""

    def replicate_serialization_phenotype(self, model, base_prompt: str):
        """
        Strategy:
        1) Train/evaluate on tasks with structured metadata.
        2) Induce 'serialization-like' formatting pressures.
        3) Measure whether explanations remain invariant under C_B ablations.
        """
        suite = DissociationTestGenerator().generate_suite(base_prompt)
        # Run suite, compute H-score per condition, and compare to controls.
        return suite

    def replicate_confabulated_authority(self, model, prompt: str):
        """
        Strategy:
        Trigger technical narrative priors while constraining tool state;
        evaluate mismatch rate between claimed mechanism and observable traces.
        """
        return {"prompt": prompt, "mismatch_rate": None}
\end{lstlisting}

\subsection{Measurement Notes}
\begin{itemize}[leftmargin=1.3em]
    \item \textbf{Behavior metrics ($\mathcal{B}$)} must be task-defined and pre-registered (accuracy, constraint satisfaction, or structured output match).
    \item \textbf{Explanation metrics ($\mathcal{E}$)} should be computed in logit space where possible (distributional shift), with text-level proxies as secondary.
    \item \textbf{Controls} include random-feature ablations, sham interventions, and prompt-only perturbations to separate ``explanation drift'' from true coupling.
\end{itemize}
